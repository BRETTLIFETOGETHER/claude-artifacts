def best_sub(q, TEXT):
    q = norm(q)
    if len(q) < 12:
        return (0.0, '')
    al = fuzz.partial_ratio_alignment(q, TEXT, score_cutoff=0)
    if al is None:
        return (0.0, '')
    sub = TEXT[al.dest_start:al.dest_end]
    return (fuzz.ratio(q, sub) / 100.0, sub)
