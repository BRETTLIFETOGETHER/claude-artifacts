def folio(chapter, num):

def track_card(name, sub, para, bullets):
    bl = "".join([f'<li>{b}</li>' for b in bullets])
    return f'''
    <div class="track-card" style="margin-bottom:0.2in;">

      <ul class="track-list">{bl}</ul>
    </div>'''
