def render_categories(cats, prefix):
    parts = []
    for i, cat in enumerate(cats):
        bg = PALETTE[i % len(PALETTE)]
        sec = category_section(cat, bg)
        sec = sec.replace(f'id="cat-{cat["roman"]}"', f'id="{prefix}-{cat["roman"]}"', 1)
        parts.append(sec)
    return "".join(parts)
