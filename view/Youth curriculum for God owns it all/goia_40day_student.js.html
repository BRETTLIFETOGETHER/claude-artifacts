const {
  Document, Packer, Paragraph, TextRun, PageBreak,
  AlignmentType, BorderStyle, WidthType, ShadingType,
  LevelFormat, Table, TableRow, TableCell
} = require('docx');
const fs = require('fs');

// ─── COLORS ───────────────────────────────────────────────
const NAVY   = "1B3A5C";
const GOLD   = "C8922A";
const TEAL   = "2A7B8C";
const LB     = "E8F0F7";   // light blue
const LG_    = "FDF3E3";   // light gold
const LT     = "E0F2F5";   // light teal
const GRAY   = "F3F4F6";
const MGRAY  = "6B7280";
const WHITE  = "FFFFFF";
const DARK   = "1F2937";

const NO = { style: BorderStyle.NONE };

// ─── PRIMITIVES ───────────────────────────────────────────
const sp  = (b=80,a=80) => ({ before:b, after:a });
const pb  = () => new Paragraph({ children:[new PageBreak()] });
const gap = (h=120) => new Paragraph({ spacing:sp(h,0), children:[] });

const t = (text, opts={}) =>
  new TextRun({ text, font:"Arial", size:22, color:DARK, ...opts });

const para = (runs, spacing=sp(60,60), align=AlignmentType.LEFT) =>
  new Paragraph({ spacing, alignment:align, children: Array.isArray(runs) ? runs : [t(runs)] });

const body = (text, before=60, after=60) =>
  para(text, sp(before,after));

const bold = (text) => para([t(text,{bold:true})], sp(60,60));

const h2 = (text) => new Paragraph({
  spacing: sp(240,80),
  children: [t(text,{bold:true,size:28,color:NAVY})]
});

const slab = (text) => new Paragraph({
  spacing: sp(220,60),
  children: [t(text.toUpperCase(),{bold:true,size:17,color:GOLD})]
});

const bul = (text) => new Paragraph({
  numbering:{reference:"bullets",level:0},
  spacing:sp(36,36),
  children:[t(text)]
});

// ─── BOXES ────────────────────────────────────────────────
const box = (children, fill=LB, left=NAVY) =>
  new Table({
    width:{size:9360,type:WidthType.DXA}, columnWidths:[140,9220],
    borders:{top:NO,bottom:NO,left:NO,right:NO,insideH:NO,insideV:NO},
    rows:[new TableRow({children:[
      new TableCell({
        width:{size:140,type:WidthType.DXA},
        shading:{fill:left,type:ShadingType.CLEAR},
        borders:{top:NO,bottom:NO,right:NO,
          left:{style:BorderStyle.SINGLE,size:24,color:left}},
        children:[new Paragraph({children:[]})]
      }),
      new TableCell({
        width:{size:9220,type:WidthType.DXA},
        shading:{fill,type:ShadingType.CLEAR},
        margins:{top:120,bottom:120,left:220,right:220},
        borders:{top:NO,bottom:NO,left:NO,right:NO},
        children: Array.isArray(children) ? children : [children]
      })
    ]})]
  });

const scripture = (verse, ref) => box([
  para([t(`"${verse}"`,{italics:true,size:22,color:NAVY})], sp(40,20)),
  para([t(`— ${ref}`,{bold:true,size:19,color:GOLD})], sp(0,40))
], LB, NAVY);

const pq = (text) => box([
  para([t(text,{bold:true,italics:true,size:23,color:NAVY})],sp(80,80))
], LG_, GOLD);

const declaration = (text) => box([
  para([t("WRITE THIS DECLARATION",{bold:true,size:17,color:GOLD})],sp(0,60)),
  para([t(`"${text}"`,{bold:true,italics:true,size:22,color:NAVY})],sp(0,20))
], LB, GOLD);

// Fill-in exercise line
const fill = (n, prompt, hint=null) => {
  const kids = [
    para([t(`${n}.  `,{bold:true,size:22,color:NAVY}), t(prompt,{size:22})], sp(40,8))
  ];
  if(hint) kids.push(para([t(hint,{italics:true,size:19,color:MGRAY})],sp(0,8)));
  kids.push(new Paragraph({
    spacing:sp(4,4),
    border:{bottom:{style:BorderStyle.SINGLE,size:2,color:"CCCCCC"}},
    children:[t("   ")]
  }));
  kids.push(new Paragraph({
    spacing:sp(4,20),
    border:{bottom:{style:BorderStyle.SINGLE,size:2,color:"CCCCCC"}},
    children:[t("   ")]
  }));
  return box(kids, GRAY, TEAL);
};

const complete = (prompt) => new Paragraph({
  spacing:sp(60,8),
  children:[
    t(`"${prompt}`,{italics:true,size:22}),
    t(" ___________________________\u201d",{size:22,color:TEAL})
  ]
});

const steps = (a,b,c) => [
  slab("Next Steps \u2014 Pick one. Do it today."),
  box([
    para([t("Option A",{bold:true,size:22,color:NAVY})],sp(0,40)),
    body(a),
    gap(40),
    para([t("Option B",{bold:true,size:22,color:NAVY})],sp(0,40)),
    body(b),
    gap(40),
    para([t("Option C",{bold:true,size:22,color:NAVY})],sp(0,40)),
    body(c)
  ], GRAY, TEAL)
];

const prayer = (text) => [
  slab("Closing Prayer"),
  box([para([t(text,{italics:true,size:22})],sp(20,20))], LT, TEAL)
];
