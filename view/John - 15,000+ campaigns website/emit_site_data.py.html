    mm = REF_RX.match(ref)
    if not mm: return False
    b, c, v1, v2 = mm.group(1), int(mm.group(2)), int(mm.group(3)), mm.group(4)
    mx = VC.get((b, c))
    if mx is None: return False
    v2 = int(v2) if v2 else v1
    return 1 <= v1 <= v2 <= mx
