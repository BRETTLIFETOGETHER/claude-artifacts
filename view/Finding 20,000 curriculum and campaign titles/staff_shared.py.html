# ===== THE EIGHT-QUARTER DEVELOPMENTAL JOURNEY (shared spine, role-specific milestones) =====
QUARTERS = [
 ["Q1","Months 1-3","Learn the room",
  "Change nothing. Meet everyone. Write down what you find before you get used to it — you only get one first ninety days and the fresh eyes are the asset.",
  ["Who are the twelve people I must know by name?","What is actually working that I should not touch?",
   "What is broken that everyone has stopped seeing?","Where is the money and who decides it?",
   "What did my predecessor do that people miss?"],
  "A written one-page assessment of the ministry as you found it, given to your supervisor."],
 ["Q2","Months 4-6","Fix one thing",
  "One visible, finishable improvement. Not a restructure. The point is not the fix, it is proving you can finish something.",
  ["What is the smallest change with the most visible benefit?","Who has to agree before I move?",
   "What will this cost in goodwill?","How will anyone know it worked?",
   "What am I choosing not to do this quarter?"],
  "One completed improvement with a before and after somebody else can see."],
 ["Q3","Months 7-9","Build the first team",
  "Stop doing it alone. Recruit and train two to five people. Most roles fail here, not from incompetence but from never handing anything over.",
  ["Who are three people I could recruit this quarter?","What is the smallest real job I can give away?",
   "What training would they actually need?","What will I stop doing when they start?",
   "Who is the one I would want to eventually replace me?"],
  "A named team with defined roles and one completed training."],
 ["Q4","Months 10-12","Write the year",
  "Plan twelve months on paper — calendar, budget, and the one outcome you will be judged on. The first annual plan is when a role becomes a ministry.",
  ["What are the three outcomes for next year?","What does this cost and who approves it?",
   "Where does this connect to what the church is already doing?","What did I learn in year one that changes the plan?",
   "What would make next year a failure?"],
  "A written annual plan with budget, calendar and three measurable outcomes."],
 ["Q5","Months 13-15","Deepen the bench",
  "Year two is a leadership year. Move from recruiting helpers to developing leaders who recruit their own.",
  ["Which team members could lead rather than help?","What am I still doing that somebody else should own?",
   "Who has plateaued and needs a harder assignment?","What training do my leaders need that I cannot give?",
   "Where is the ministry dependent on me personally?"],
  "Two people leading something you used to lead."],
 ["Q6","Months 16-18","Systematize",
  "Write down what only lives in your head. If you were hit by a bus, how much of this ministry survives? That number is your actual health score.",
  ["What is undocumented that matters?","Which processes fail when I am away?",
   "What would a new person need on day one?","Where are we improvising the same thing every month?",
   "What can be templated rather than decided each time?"],
  "A written operations document a successor could run from."],
 ["Q7","Months 19-21","Measure honestly",
  "Look at whether it worked. Not activity — outcomes. This is the quarter most people skip, and skipping it is why ministries repeat bad years.",
  ["What did we say we would accomplish?","What actually changed in people?",
   "What did we do that produced nothing?","What is the one number that matters most here?",
   "What would I stop if I were new?"],
  "An honest annual review including what failed, presented to your supervisor."],
 ["Q8","Months 22-24","Multiply or hand off",
  "Either this ministry produces its next leader or it depends on you permanently. Year two ends by proving which.",
  ["Who could do most of my job now?","What am I uniquely required for, honestly?",
   "What would I need to give away to grow?","Where does this ministry go next and who takes it?",
   "Am I building something or maintaining something?"],
  "One person trained to take a defined part of your role, permanently."],
]

GAP_METHOD = [
 ["Name the outcome","Not the activity. 'Twelve trained group leaders by June' rather than 'improve group ministry.'"],
 ["Score where you are","One to five, honestly, on the competency map for your level. Guessing high wastes the quarter."],
 ["Find the largest gap","The biggest distance between current and required, not the one that is most comfortable to work on."],
 ["Name one action","One. Quarterly plans with six priorities have none."],
 ["Name the evidence","What will exist at the end of the quarter that does not exist now, that another person could look at."],
 ["Book the review","A date on a calendar with a named person. A gap analysis nobody reviews is a diary entry."],
]

VOL_BY_SIZE = [
 ["Under 100","One person recruits everyone personally. No system yet, and that is correct — do not build infrastructure for a problem you do not have. One written role description per job is enough."],
 ["100 to 250","Recruitment must move from the pastor to team leaders. This is the transition most churches miss, and it is why growth stalls here."],
 ["250 to 500","Systems replace memory. Screening, onboarding and scheduling need to run without anyone holding it in their head."],
 ["500 to 1,000","Team leaders lead team leaders. The staff member develops the layer above the volunteers rather than the volunteers themselves."],
 ["Over 1,000","A pipeline, not a recruitment drive. Every ministry runs its own funnel and the central function is training and standards."],
]

VOL_BY_SEASON = [
 ["Fall launch","The largest recruiting window of the year. Ask in August, onboard in September, and expect the highest yield you will see."],
 ["January","The second window. People make commitments in January that they will not make in March."],
 ["Easter","High attendance, high need, low commitment. Recruit for the day, then follow up in the two weeks after with a longer ask."],
 ["Summer","Coverage season. Do not recruit new volunteers; protect the ones you have with rotation and rest."],
 ["Christmas","Short, intense, and the best on-ramp for a first-time server. One-off serving with no ongoing obligation converts remarkably well."],
]

COACH_CADENCE = [
 ["Weekly","30 minutes","Supervisor and staff member. What happened, what is stuck, what do you need. Never a status report — a status report can be an email."],
 ["Monthly","90 minutes","Coach and staff member. One competency, one real situation, one assignment. The coach is not the supervisor and should not be."],
