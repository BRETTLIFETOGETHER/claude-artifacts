WFI_THESIS = ("A pastor has forty-eight hours, a family he may not know, and a room where half the people have no "
 "category for what is happening. Intelligence is asking the right questions fast enough that the service fits "
 "this family rather than the last one.")

WFI_FLOW = [
 ["01","Intake","The first phone call. Fifteen questions that determine everything else, asked while the family is still in shock and answering plainly."],
 ["02","Situation profile","Type of death or wedding, family shape, faith of the family, the room you will face, and the constraints."],
 ["03","The gap","What this family needs that a standard service will not give them. This is the whole product."],
 ["04","Service assembly","The right category, the right entries, the right tools, the right add-ins, generated against the profile."],
 ["05","The family conversation","One meeting with the questions written out. Most pastors improvise this and lose material they cannot recover."],
 ["06","Delivery and follow-up","The service, then the follow-up schedule that begins the same week."],
]
